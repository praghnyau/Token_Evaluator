\section{Project Overview}
\begin{frame}{Project Overview}
  \begin{block}{Objective}
    Build a \textbf{real-time Prompt Analytics Assistant} that monitors AI prompts,
    predicts token usage, estimates cost, and classifies prompt quality — live, as the user types.
  \end{block}
  \vspace{0.4em}
  \begin{columns}
    \column{0.5\textwidth}
    \textbf{Target Platforms}
    \begin{itemize}
      \item Google Gemini
      \item ChatGPT
      \item Claude AI
      \item Antigravity CLI
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Core Value}
    \begin{itemize}
      \item Reduce wasted tokens
      \item Improve prompt quality
      \item Track API cost in real time
      \item Learn from usage history
    \end{itemize}
  \end{columns}
\end{frame}

\section{ML Pipeline}
\begin{frame}{ML Pipeline}
  \begin{block}{Training Data}
    \textbf{Team\_6\_Gemini\_dataset.csv} — $\sim$520K rows \quad|\quad
    \textbf{evaluation.csv} — $\sim$170K rows (labelled: Good / Average / Bad)
  \end{block}
  \vspace{0.4em}
  \begin{columns}
    \column{0.33\textwidth}
    \centering
    \textbf{Quality Classifier}\\[4pt]
    TF-IDF (500 features)\\
    + structural features\\
    (length, word count,\\
    code blocks, lists)\\[4pt]
    Logistic Regression\\
    \textit{or} Random Forest\\
    (best CV score wins)
    \column{0.33\textwidth}
    \centering
    \textbf{Token Regressor}\\[4pt]
    Ridge Regression\\
    on TF-IDF + model type\\
    + input token count\\[4pt]
    Falls back to\\
    $2.5\times$ input tokens\\
    if DB is sparse
    \column{0.33\textwidth}
    \centering
    \textbf{Similarity Search}\\[4pt]
    SentenceTransformers\\
    \texttt{all-MiniLM-L6-v2}\\
    + FAISS IndexFlatIP\\[4pt]
    TF-IDF cosine\\
    similarity fallback\\
    (top-3 results)
  \end{columns}
\end{frame}

\section{Key Challenges \& Solutions}
\begin{frame}{Key Challenges \& Solutions}
  \begin{table}
    \small
    \begin{tabular}{p{4.5cm} p{6.5cm}}
      \toprule
      \textbf{Challenge} & \textbf{Solution} \\
      \midrule
      Cold-start (empty DB) & Seed DB from 520K-row CSV on first run \\
      Single-class training edge case & Skip classifier training; use majority class fallback \\
      Gemini API latency in real-time & 4s timeout + offline ML fallback \\
      Token counting for Gemini & \texttt{tiktoken} cl100k\_base (close approximation) \\
      Stale model after new feedback & Background retrain task after every feedback/run \\
      Schema migration (cost columns) & \texttt{PRAGMA table\_info} + \texttt{ALTER TABLE} on startup \\
      \bottomrule
    \end{tabular}
  \end{table}
\end{frame}
